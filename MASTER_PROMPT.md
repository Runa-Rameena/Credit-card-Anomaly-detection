### Step 4.3 — Create the DAGs

#### DAG 1: `airflow/dags/pipeline_health_dag.py`

**Purpose:** Runs every 5 minutes. Checks:
1. Is MongoDB reachable and has at least 1 document?
2. Is the Kafka topic `transactions` reachable?
3. Counts anomalies in the last 5 minutes and logs the result.

```python
from airflow import DAG
from airflow.operators.python import PythonOperator
from datetime import datetime, timedelta
import logging

logger = logging.getLogger(__name__)

default_args = {
    'owner': 'pipeline',
    'retries': 1,
    'retry_delay': timedelta(minutes=1),
    'email_on_failure': False,
}

def check_mongodb(**context):
    from pymongo import MongoClient
    import os
    from dotenv import load_dotenv
    load_dotenv(os.path.expanduser('~/projects/credit_card_anomaly/config/.env'))
    uri = os.getenv('MONGO_URI', 'mongodb://127.0.0.1:27017')
    client = MongoClient(uri, serverSelectionTimeoutMS=5000)
    db = client['credit_card_db']
    count = db['transactions'].count_documents({})
    anoms = db['transactions'].count_documents({'rule_anomaly': True})
    logger.info(f"[HealthCheck] MongoDB OK — Total: {count}, Anomalies: {anoms}")
    context['ti'].xcom_push(key='mongo_count', value=count)
    context['ti'].xcom_push(key='mongo_anomalies', value=anoms)
    client.close()

def check_recent_anomalies(**context):
    from pymongo import MongoClient
    from datetime import datetime, timedelta, timezone
    import os
    from dotenv import load_dotenv
    load_dotenv(os.path.expanduser('~/projects/credit_card_anomaly/config/.env'))
    uri = os.getenv('MONGO_URI', 'mongodb://127.0.0.1:27017')
    client = MongoClient(uri, serverSelectionTimeoutMS=5000)
    db = client['credit_card_db']
    cutoff = datetime.now(timezone.utc) - timedelta(minutes=5)
    recent_anoms = db['transactions'].count_documents({
        'rule_anomaly': True,
        'processing_time': {'$gte': cutoff}
    })
    logger.info(f"[HealthCheck] Anomalies in last 5 mins: {recent_anoms}")
    if recent_anoms > 10:
        logger.warning(f"[ALERT] High anomaly rate detected: {recent_anoms} in 5 mins!")
    client.close()

with DAG(
    dag_id='pipeline_health_check',
    default_args=default_args,
    description='Checks MongoDB health and anomaly rate every 5 minutes',
    schedule='*/5 * * * *',
    start_date=datetime(2024, 1, 1),
    catchup=False,
    tags=['health', 'monitoring'],
) as dag:
    check_mongo_task = PythonOperator(
        task_id='check_mongodb_health',
        python_callable=check_mongodb,
    )
    check_anomaly_task = PythonOperator(
        task_id='check_recent_anomalies',
        python_callable=check_recent_anomalies,
    )
    check_mongo_task >> check_anomaly_task
```

#### DAG 3: `airflow/dags/data_cleanup_dag.py`

**Purpose:** Runs daily at 3 AM. Archives MongoDB documents older than 7 days into a separate `transactions_archive` collection and deletes them from `transactions`.

```python
from airflow import DAG
from airflow.operators.python import PythonOperator
from datetime import datetime, timedelta
import logging

logger = logging.getLogger(__name__)

default_args = {
    'owner': 'pipeline',
    'retries': 1,
    'retry_delay': timedelta(minutes=5),
    'email_on_failure': False,
}

def archive_old_transactions(**context):
    from pymongo import MongoClient
    from datetime import datetime, timedelta, timezone
    import os
    from dotenv import load_dotenv
    load_dotenv(os.path.expanduser('~/projects/credit_card_anomaly/config/.env'))
    uri = os.getenv('MONGO_URI', 'mongodb://127.0.0.1:27017')
    client = MongoClient(uri, serverSelectionTimeoutMS=5000)
    db = client['credit_card_db']
    
    cutoff = datetime.now(timezone.utc) - timedelta(days=7)
    old_docs = list(db['transactions'].find({'processing_time': {'$lt': cutoff}}))
    
    if old_docs:
        db['transactions_archive'].insert_many(old_docs)
        result = db['transactions'].delete_many({'processing_time': {'$lt': cutoff}})
        logger.info(f"Archived {result.deleted_count} old documents to transactions_archive.")
    else:
        logger.info("No old documents to archive.")
    client.close()

with DAG(
    dag_id='daily_data_cleanup',
    default_args=default_args,
    description='Archives MongoDB docs older than 7 days daily at 3 AM',
    schedule='0 3 * * *',
    start_date=datetime(2024, 1, 1),
    catchup=False,
    tags=['cleanup', 'mongodb'],
) as dag:
    cleanup_task = PythonOperator(
        task_id='archive_old_transactions',
        python_callable=archive_old_transactions,
    )
```

## PHASE 9 — Error Prevention Rules

These must be enforced throughout the code:

1. **Always load `.env` before any `os.getenv()` call.**
2. **Never hardcode IPs.** Use env vars with `localhost` as the default.
3. **Spark JAR versions must exactly match** `pyspark` version (Scala suffix `2.12` vs `2.13` too).
4. **Pandas UDF requires `pyarrow`** — it must be installed in the venv.
5. **Airflow `AIRFLOW_HOME` must be set** before any `airflow` command.
6. **MongoDB write errors in `foreachBatch` must be caught** — never let them crash the stream query.
7. **`start.sh` must be executable:** `chmod +x scripts/start.sh scripts/stop.sh`
8. **All Python files must have `__init__.py`** in their directories for module imports to work.
9. **Model `.pkl` files must exist** before the streaming job starts — `start.sh` checks and trains if missing.
10. **For Laptop 2, the pyspark version must exactly match Laptop 1** — enforce this in `start.sh worker` mode.
